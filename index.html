// Lógica de Registro de Nuevos Solicitantes mediante función interna definitiva document.getElementById('formRegistro').addEventListener('submit', async (e) => { e.preventDefault(); errContainer.classList.add('hidden'); successContainer.classList.add('hidden'); const email = document.getElementById('regEmail').value.trim().toLowerCase(); const password = document.getElementById('regPassword').value; const nombre_completo = document.getElementById('regNombre').value.trim(); const btn = document.getElementById('btnRegSubmit'); btn.innerHTML = ' Registrando...'; btn.disabled = true; try { // Llamada a la función RPC de Supabase (omite restricciones de API Auth) const { data, error } = await window.supabaseClient.rpc('registrar_solicitante', { p_email: email, p_password: password, p_nombre_completo: nombre_completo }); if (error) throw error; if (data && data.success === false) throw new Error(data.error); successTexto.innerText = "¡Cuenta creada con éxito! Ya puedes iniciar sesión."; successContainer.classList.remove('hidden'); document.getElementById('formRegistro').reset(); setTimeout(() => cambiarTab('login'), 3000); } catch (err) { errTexto.innerText = "Error en el registro: " + err.message; errContainer.classList.remove('hidden'); } finally { btn.disabled = false; btn.innerHTML = 'Registrarme como Solicitante'; } });
