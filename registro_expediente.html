Crear tabla de expedientes CREATE TABLE IF NOT EXISTS public.expedientes ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), codigo_expediente TEXT UNIQUE NOT NULL, titulo_obra TEXT NOT NULL, area_senadi TEXT NOT NULL, tipo_obra TEXT NOT NULL, autor_nombre TEXT NOT NULL, autor_cedula TEXT NOT NULL, autor_telefono TEXT NOT NULL, archivo_cedula_url TEXT, archivo_obra_url TEXT, contrato_url TEXT, user_id UUID REFERENCES auth.users(id) ON DELETE SET NULL, created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL ); -- 2. Crear tabla de autores y coautores CREATE TABLE IF NOT EXISTS public.autores ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), expediente_id UUID REFERENCES public.expedientes(id) ON DELETE CASCADE, nombre_completo TEXT NOT NULL, cedula TEXT NOT NULL, telefono TEXT, tipo TEXT DEFAULT 'coautor', -- 'principal' o 'coautor' created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL ); -- 3. Habilitar Seguridad (RLS) ALTER TABLE public.expedientes ENABLE ROW LEVEL SECURITY; ALTER TABLE public.autores ENABLE ROW LEVEL SECURITY; -- 4. Políticas de acceso para expedientes y autores CREATE POLICY "Lectura pública de expedientes autenticados" ON public.expedientes FOR SELECT USING (true); CREATE POLICY "Inserción de expedientes autenticados" ON public.expedientes FOR INSERT WITH CHECK (true); CREATE POLICY "Actualización de expedientes autenticados" ON public.expedientes FOR UPDATE USING (true); CREATE POLICY "Lectura pública de autores" ON public.autores FOR SELECT USING (true); CREATE POLICY "Inserción de autores" ON public.autores FOR INSERT WITH CHECK (true);
